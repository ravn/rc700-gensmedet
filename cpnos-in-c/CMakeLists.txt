cmake_minimum_required(VERSION 3.20)
project(cpnos-in-c C)

# This CMakeLists.txt exists for CLion indexing and navigation only.
# The actual build uses Makefiles with clang (--target=z80) or z88dk-zsdcc.
# Z80-specific asm is guarded by #ifdef __z80__ (only defined by llvm-z80).

# z88dk headers (stdint.h, etc.) — from parent or standalone open
if(NOT DEFINED Z88DK_INCLUDE)
    get_filename_component(Z88DK_INCLUDE
        "${CMAKE_CURRENT_SOURCE_DIR}/../../z88dk/include" ABSOLUTE)
endif()

# Canonicalise the `..` so clangd/CLion see an absolute path with no
# parent-dir traversal — some tooling caches paths verbatim and chokes
# on the unresolved form.
get_filename_component(SHARED_INCLUDE
    "${CMAKE_CURRENT_SOURCE_DIR}/../cpnos-shared/include" ABSOLUTE)

add_library(cpnos_sources OBJECT EXCLUDE_FROM_ALL
    src/cpnos_main.c
    src/hal.h
    src/init.c
    src/resident.c
    src/snios_c.c
    src/transport.h
    src/transport_pio.c
    src/transport_sio.c
)

# cpnos_buildinfo.h and cpnos_addrs.h are GENERATED by the Makefile
# into the per-compiler build dir (clang/ or sdcc/, also named
# clang-prom1lineprog/ for the production line-program build).  Real
# builds add `-I$(BUILDDIR)` so the compiler finds them.  For CLion
# indexing we point at the clang/ build dir first (the production
# COMPILER default) and fall back to a generated placeholder in src/
# when nothing is built yet.
set(CPNOS_BUILDINFO_PLACEHOLDER ${CMAKE_CURRENT_SOURCE_DIR}/src/cpnos_buildinfo.h)
if(NOT EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/clang/cpnos_buildinfo.h AND
   NOT EXISTS ${CPNOS_BUILDINFO_PLACEHOLDER})
    file(WRITE ${CPNOS_BUILDINFO_PLACEHOLDER}
        "#ifndef CPNOS_BUILDINFO_H\n"
        "#define CPNOS_BUILDINFO_H\n"
        "/* CLion-placeholder — real values are generated into clang/ at build time. */\n"
        "#define BUILD_DATE_STR \"CLion-placeholder\"\n"
        "#define GIT_HASH_STR   \"CLion-placeholder\"\n"
        "#define BUILD_INFO_STR BUILD_DATE_STR \" \" GIT_HASH_STR\n"
        "#endif\n")
endif()
set(CPNOS_ADDRS_PLACEHOLDER ${CMAKE_CURRENT_SOURCE_DIR}/src/cpnos_addrs.h)
if(NOT EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/clang/cpnos_addrs.h AND
   NOT EXISTS ${CPNOS_ADDRS_PLACEHOLDER})
    file(WRITE ${CPNOS_ADDRS_PLACEHOLDER}
        "#ifndef CPNOS_ADDRS_H\n"
        "#define CPNOS_ADDRS_H\n"
        "/* CLion-placeholder — real values are extracted from the link map. */\n"
        "#define CPNOS_BDOS_ADDR   0xE800\n"
        "#define CPNOS_NDOS_ADDR   0xDE80\n"
        "#define CPNOS_NDOSE_ADDR  0xDE83\n"
        "#define CPNOS_NDOSRL_ADDR 0xDA80\n"
        "#define CPNOS_TPA_KB      55\n"
        "#endif\n")
endif()

target_include_directories(cpnos_sources PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}          # root: for #include "compiler/compat.h"
    ${CMAKE_CURRENT_SOURCE_DIR}/clang
    ${CMAKE_CURRENT_SOURCE_DIR}/src
    ${SHARED_INCLUDE}                    # cpnos-shared/include: for cfgtbl.h
    ${Z88DK_INCLUDE}
)

# Belt-and-braces: also link the cpnos_shared_headers INTERFACE library
# when it's available (i.e. when the parent CMakeLists.txt loaded
# cpnos-shared's CMakeLists.txt before this one).  Propagates the
# canonical INTERFACE include path via CMake's standard mechanism, so
# CLion's clangd sees the same path the SHARED_INCLUDE -I gives.
if(TARGET cpnos_shared_headers)
    target_link_libraries(cpnos_sources PRIVATE cpnos_shared_headers)
endif()

# Defines mirror the Makefile's production clang build (lines 115-122 + 217-219).
# Values match the default targets so the indexer sees the same #ifdef branches
# the production build does.
target_compile_definitions(cpnos_sources PRIVATE
    ENABLE_FDC=0
    MIRROR_SIOB=0
    BOOT_MARK_ENABLED=1
    RC702_SLAVEID=0x01
    PIO_SPEED_TEST=0
    PIO_LOOPBACK_TEST
    TRANSPORT_NAME="pio"
)

# Custom targets that invoke the real Makefile
add_custom_target(cpnos-prom1-clang
    COMMAND make prom1-lineprog COMPILER=clang
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    COMMENT "Build cpnos-in-c PROM1-only line program (clang)"
)

add_custom_target(cpnos-prom1-sdcc
    COMMAND make prom1-lineprog COMPILER=sdcc
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    COMMENT "Build cpnos-in-c PROM1-only line program (sdcc)"
)

add_custom_target(cpnos-polypascal-test
    COMMAND make polypascal-test
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    COMMENT "Run cpnos polypascal-test in MAME"
)
